\project{Sea Anchor}{\href{https://fishing-grounds-recsys.onrender.com/}{live app}\,$\cdot$\,\href{https://github.com/Vanamali-Sims/fishing-grounds-recsys}{code}}{Python, DuckDB SQL, Parquet, implicit ALS, FastAPI, React}

\project{Melbourne Footfall}{\href{https://github.com/Vanamali-Sims/howMelbMoves}{code}\,$\cdot$\,\href{https://public.tableau.com/app/profile/sai.vanamali.simhambhatla4623/viz/MelbourneFootfall/Dashboard3}{Tableau dashboard}}{SQL, dbt, DuckDB, Python, LightGBM, GitHub Actions, Tableau}

  \item LightGBM beats a seasonal-naive baseline (MASE 0.87 vs 0.98 over 89 rolling-origin folds), feeding Tableau directly.
\end{itemize}
